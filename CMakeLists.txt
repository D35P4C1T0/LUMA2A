cmake_minimum_required(VERSION 3.22)
project(Luma2A VERSION 0.1.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_SOURCE_DIR "" CACHE PATH "Local JUCE checkout (optional)")
if(JUCE_SOURCE_DIR)
  add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE")
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
endif()
if(CMAKE_CROSSCOMPILING)
  set(OPTO_AUTO_MANIFEST FALSE)
else()
  set(OPTO_AUTO_MANIFEST TRUE)
endif()
set(OPTO_FORMATS VST3 Standalone)
if(APPLE)
  list(APPEND OPTO_FORMATS AU)
endif()
juce_add_plugin(Luma2A
  COMPANY_NAME "Luma2A" BUNDLE_ID "audio.optoa2a.plugin"
  PLUGIN_MANUFACTURER_CODE OpTo PLUGIN_CODE OA2a
  FORMATS ${OPTO_FORMATS} PRODUCT_NAME "Luma2A"
  IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
  COPY_PLUGIN_AFTER_BUILD FALSE VST3_AUTO_MANIFEST ${OPTO_AUTO_MANIFEST})
target_sources(Luma2A PRIVATE Source/Processor.cpp Source/Editor.cpp)
target_include_directories(Luma2A PUBLIC Source)
target_compile_definitions(Luma2A PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Luma2A PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
juce_add_console_app(UiPreview PRODUCT_NAME "UiPreview")
target_sources(UiPreview PRIVATE tools/UiPreview.cpp)
target_compile_definitions(UiPreview PRIVATE JUCE_PLUGINHOST_VST3=1)
target_link_libraries(UiPreview PRIVATE Luma2A juce::juce_audio_utils)

add_executable(model_render tools/model_render.cpp)
target_include_directories(model_render PRIVATE Source)
if(MSVC)
  target_compile_options(Luma2A PRIVATE /fp:strict)
  target_compile_options(model_render PRIVATE /fp:strict)
else()
  target_compile_options(Luma2A PRIVATE -ffp-contract=off -fno-fast-math)
  target_compile_options(model_render PRIVATE -ffp-contract=off -fno-fast-math)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(ModelChecks tools/model_checks.cpp)
  target_include_directories(ModelChecks PRIVATE Source)
  if(MSVC)
    target_compile_options(ModelChecks PRIVATE /fp:strict)
  else()
    target_compile_options(ModelChecks PRIVATE -ffp-contract=off -fno-fast-math)
  endif()
  add_test(NAME ModelChecks COMMAND ModelChecks)
  if(NOT CMAKE_CROSSCOMPILING)
    add_test(NAME PluginStateChecks COMMAND UiPreview --self-test)
  endif()
endif()

if(APPLE)
  foreach(bundleTarget Luma2A_VST3 Luma2A_AU Luma2A_Standalone)
    add_custom_command(TARGET ${bundleTarget} POST_BUILD COMMAND codesign --force --deep --sign - "$<TARGET_BUNDLE_DIR:${bundleTarget}>" VERBATIM)
  endforeach()
endif()
